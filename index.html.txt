    .days { font-weight: bold; color: #d32f2f; }
    input { width: 100%; padding: 8px; margin: 5px 0; border: 1px solid #ddd; border-radius: 5px; }
    button { background: #00c853; color: white; border: none; padding: 10px; border-radius: 5px; width: 100%; font-size: 16px; margin-top: 5px; }

<body>
  <div class="header">
    <h1>🔥 Товары с истекающим сроком</h1>
    <p>Обновите остатки на складе и полках</p>
  </div>
  <div id="items-container"></div>

  <script>
    const SCRIPT_URL = "https://script.google.com/macros/s/AKfycbxZhaAd0kU7QNOX5ln_NS8XPwErY9pANITisotdeSHBhgD3wB9J3oxrtGfSMChkO57O/exec"; // ← ВСТАВЬТЕ СВОЙ URL ЗДЕСЬ!

    async function loadItems() {
      const container = document.getElementById('items-container');
      container.innerHTML = '<div class="item">Загрузка...</div>';
      
      try {
        const response = await fetch(`${SCRIPT_URL}?action=getItems`);
        const items = await response.json();
        
        if (!Array.isArray(items)) {
          container.innerHTML = `<div class="item">❌ Ошибка: ${JSON.stringify(items)}</div>`;
          return;
        }
        
        const urgentItems = items.filter(item => item.days <= 30 && item.quantity > 0);
        
        if (urgentItems.length === 0) {
          container.innerHTML = '<div class="item">✅ Все сроки в норме!</div>';
          return;
        }
        
        container.innerHTML = '';
        urgentItems.forEach(item => {
          const itemDiv = document.createElement('div');
          itemDiv.className = `item ${item.days <= 7 ? 'expired' : 'warning'}`;
          
          itemDiv.innerHTML = `
            <h3>${item.name} (арт. ${item.article})</h3>
            <div>Срок: ${item.expiry} <span class="days">(${item.days} дн.)</span></div>
            <div>Остаток в таблице: ${item.quantity} шт.</div>
            <input type="number" min="0" placeholder="Фактический остаток" 
                   data-article="${item.article}" class="actual-qty">
            <button onclick="updateStock(this)" data-article="${item.article}">Сохранить</button>
            <div class="updated" id="status-${item.article}"></div>
          `;
          container.appendChild(itemDiv);
        });
      } catch (err) {
        container.innerHTML = `<div class="item">❌ Ошибка загрузки: ${err.message}</div>`;
      }
    }

    async function updateStock(button) {
      const article = button.getAttribute('data-article');
      const input = document.querySelector(`input[data-article="${article}"]`);
      const statusEl = document.getElementById(`status-${article}`);
      
      const actualQty = parseInt(input.value);
      if (isNaN(actualQty) || actualQty < 0) {
        statusEl.textContent = "Введите число ≥ 0";
        return;
      }
      
      statusEl.textContent = "Сохраняю...";
      
      try {
        const response = await fetch(`${SCRIPT_URL}?action=updateStock`, {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify({ article, actualQty })
        });
        
        const result = await response.json();
        
        statusEl.textContent = result.success 
          ? "✅ Сохранено!" 
          : `❌ Ошибка: ${result.error || 'неизвестно'}`;
        
        if (result.success) {
          setTimeout(() => {
            input.value = '';
            statusEl.textContent = '';
          }, 1500);
        }
      } catch (err) {
        statusEl.textContent = `❌ Ошибка: ${err.message}`;
      }
    }

    window.onload = loadItems;
  </script>
